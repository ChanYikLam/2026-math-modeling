```python
# model_q1.py
import numpy as np
from scipy.optimize import linprog
from scipy.sparse import lil_matrix
from config import CFG

def build_q1(L, PV, P, cfg=CFG, curtail=True, periodic=True):
    """问题1 单日 LP。
    ★★ 单位口径（务必统一，否则结果差 6 倍且求解器不报错）★★
       变量 E, c, d, s 全部是【电量 kWh】。因此：
         · 目标系数直接用 p（不是 p·Δt）
         · 功率上限是 P_MAX·Δt = 5000/6 = 833.333 kWh per slot
       平衡(电量):  E + PV·Δt + d = L·Δt + c + s
       SOC 递推:    S_t = S_{t-1} + η_c·c_t − d_t/η_d      ← 放电效率在【分母】
    变量顺序: [E(144) | c(144) | d(144) | s(144) | SOC(144)]
    """
    T, dt = cfg.T, cfg.DT
    PWR = cfg.P_MAX*dt                       # 每个时段的电量上限 (kWh)
    n = 5*T
    iE, iC, iD, iS, iSOC = 0, T, 2*T, 3*T, 4*T
    c_obj = np.zeros(n); c_obj[iE:iE+T] = P          # min Σ p·E  (E 已是 kWh)

    # 2T 个 SOC 递推行 + 1 个日周期行
    A = lil_matrix((2*T + 1, n)); b = np.zeros(2*T + 1)
    for t in range(T):
        # 电量平衡
        A[t, iE+t] = 1.0; A[t, iC+t] = -1.0; A[t, iD+t] = 1.0; A[t, iS+t] = -1.0
        b[t] = (L[t] - PV[t])*dt
        # SOC 递推:  η_c·c_t − d_t/η_d − S_t + S_{t-1} = 0
        r = T + t
        A[r, iC+t] = cfg.ETA_C; A[r, iD+t] = -1.0/cfg.ETA_D; A[r, iSOC+t] = -1.0
        if t == 0: b[r] = -cfg.SOC_INIT
        else:      A[r, iSOC+t-1] = 1.0
    if periodic:                                  # S_{T-1} = S_init（追加行，勿覆盖递推行）
        A[2*T, iSOC+T-1] = 1.0; b[2*T] = cfg.SOC_INIT

    surplus = np.maximum(PV - L, 0.0)*dt          # ★ 弃光上界 = 盈余【电量】，非盈余时段为 0
    bounds = ([(0, None)]*T + [(0, PWR)]*T + [(0, PWR)]*T
              + ([(0, float(x)) for x in surplus] if curtail else [(0, 0)]*T)
              + [(cfg.SOC_LO, cfg.SOC_HI)]*T)

    res = linprog(c_obj, A_eq=A.tocsr(), b_eq=b, bounds=bounds, method="highs")
    assert res.status == 0, res.message
    x = res.x
    return dict(cost=float(res.fun), E=x[iE:iE+T], c=x[iC:iC+T], d=x[iD:iD+T],
                s=x[iS:iS+T], SOC=x[iSOC:iSOC+T],
                dual_eq=res.eqlin.marginals, A=A, b=b, obj=c_obj)
```

```python
def build_arbitrage(P, cfg=CFG, cfg_cyclic=True):
    """★ 纯套利子问题（阈值定理的验证载体，变量仅 c,d,S 共 3T 个）★
    目标: max Σ p·(d − c)   ⇔   min Σ p·(c − d)
    约束: SOC 递推 + 日周期 + 容量界 + 功率界
    ★★ 单位：c,d 为【电量 kWh】；目标系数用 p（不乘 Δt）；功率界 = P_MAX·Δt ★★
    SOC 递推:  S_t = S_{t-1} + η_c·c_t − d_t/η_d      ← 放电效率在【分母】
    已复算（附件1 平均日）: 收益 11769.310000 元；充 26500、放 21465（比恰为 0.81）。
    ★ 等效循环的两种口径必须分开写，否则数字对不上：
       (a) **等效循环 = 总放电量 / 可用容量** = 21465 / 9600 = **2.236 次**（推荐主口径）；
       (b) 限制"单/双循环"时的**约束值** = 总放电量固定为 8 640 / 17 280 kWh
           （即 0.9× / 1.8× 可用容量，8 640 = η_c×9600）→ 收益 7256.20 / 11321.04 元。
       ⚠️ 旧文档写"2.484375 次"= η_c·Σc/9600 = 23850/9600，与 (a) 不同口径，**不要再混用**。
    """
    T, dt = cfg.T, cfg.DT
    PWR = cfg.P_MAX*dt                       # kWh per slot
    n = 3*T; iC, iD, iS = 0, T, 2*T
    obj = np.zeros(n); obj[iC:iC+T] = P; obj[iD:iD+T] = -P

    A = lil_matrix((T+1, n)); b = np.zeros(T+1)
    for t in range(T):
        A[t, iC+t] = cfg.ETA_C; A[t, iD+t] = -1.0/cfg.ETA_D; A[t, iS+t] = -1.0
        if t == 0: b[t] = -cfg.SOC_INIT
        else:      A[t, iS+t-1] = 1.0
    if cfg_cyclic:                            # 日周期（追加一行，勿覆盖递推行）
        A[T, iS+T-1] = 1.0; b[T] = cfg.SOC_INIT

    lb = np.r_[np.zeros(2*T), np.full(T, cfg.SOC_LO)]
    ub = np.r_[np.full(2*T, PWR), np.full(T, cfg.SOC_HI)]
    res = linprog(obj, A_eq=A.tocsr(), b_eq=b,
                  bounds=list(zip(lb, ub)), method="highs")
    assert res.status == 0, res.message
    x = res.x
    return dict(rev=-float(res.fun), c=x[iC:iC+T], d=x[iD:iD+T], SOC=x[iS:iS+T],
                mu=res.eqlin.marginals[:T], obj=obj, A=A, duals=res.eqlin.marginals)


def verify_threshold_theorem(arb, P, cfg=CFG, tol=1e-6):
    """★ 阈值定理的最优性证书（本文件最重要的一段）★
    由 Lagrange 平稳性（A_eq x = b 的乘子记作 mu）：
        z^c_t = p_t − η_c·μ_t          (充电变量的界乘子)
        z^d_t = −p_t + μ_t/η_d         (放电变量的界乘子)
    定理：
        满充 ⟺ z^c < 0 ；不充 ⟺ z^c > 0 ；内部 ⟺ z^c = 0
        满放 ⟺ z^d < 0 ；不放 ⟺ z^d > 0 ；内部 ⟺ z^d = 0
    等价的阈值形式：p_t < η_c·μ_t ⟹ 充电 ； p_t > μ_t/η_d ⟹ 放电
    已复算：四条满足率 100%，两条恒等式偏差 ≤2.2e-16，
            阈值四边极值均 ≈0（恰好相接，无缝隙无重叠）。
    ⚠ 对偶符号务必用最小算例标定，不要猜：
        min Σ p·E  s.t.  E ≥ max(N,0)·Δt   →   其对偶最优恰为 p。
    """
    T, dt = cfg.T, cfg.DT
    PWR = cfg.P_MAX*dt
    c, d, mu = arb["c"], arb["d"], arb["mu"]
    # 界乘子 z = obj − A^T·(等式乘子)；A 的最后一行是日周期约束
    grad = arb["obj"] - arb["A"].toarray().T @ arb["duals"]

    zc, zd = grad[:T], grad[T:2*T]
    ch_full, ch_zero = c >= PWR - tol, c <= tol
    di_full, di_zero = d >= PWR - tol, d <= tol

    checks = {
        "满充→z^c<0": bool((zc[ch_full] <  tol).all()),
        "不充→z^c>0": bool((zc[ch_zero] > -tol).all()),
        "满放→z^d<0": bool((zd[di_full] <  tol).all()),
        "不放→z^d>0": bool((zd[di_zero] > -tol).all()),
    }
    ident = {
        "z^c ≡ p − η_c·μ": float(np.abs(zc - (P - cfg.ETA_C*mu)).max()),
        "z^d ≡ −p + μ/η_d": float(np.abs(zd - (-P + mu/cfg.ETA_D)).max()),
    }
    # 阈值在边界处恰好相接（四边极值均应为 0）
    edge = {
        "充电侧上确界 max(p−η_c·μ)": float((P - cfg.ETA_C*mu)[c > 1e-9].max()),
        "非充侧下确界 min(p−η_c·μ)": float((P - cfg.ETA_C*mu)[c <= 1e-9].min()),
        "放电侧下确界 min(p−μ/η_d)": float((P - mu/cfg.ETA_D)[d > 1e-9].min()),
        "非放侧上确界 max(p−μ/η_d)": float((P - mu/cfg.ETA_D)[d <= 1e-9].max()),
    }
    lo = np.r_[np.zeros(2*T), np.full(T, cfg.SOC_LO)]
    xv = np.r_[c, d, arb["SOC"]]
    return dict(checks=checks, identities=ident, edges=edge,
                all_pass=all(checks.values()) and max(ident.values()) < 1e-10,
                improvable=int(((xv <= lo + 1e-7) & (grad < -1e-6)).sum()),
                mu_unique=int(len(np.unique(np.round(mu, 4)))),
                mu_range=(float(mu.min()), float(mu.max())))
```

## §4 最优性证书（六项检查，**任一失败即抛异常**）

```python
# certify.py
import numpy as np
from config import CFG

def certify(sol, L, PV, P, cfg=CFG, name="", tol=1e-6,
            g=None, periodic=True, soc_init=None):
    """通用审计。

    g        : 紧急购电量（kWh，1-D 长度 T）。问题 2/3/4 **必须传**；
               问题 1 传 None（等价于 g≡0）。⚠️ 漏掉 g 会把合法结算判为失败
               （实测 g[42]=600 kWh 时残差 = 600 kWh，超 1e-6 容差 6.0e8 倍）。
    periodic : 问题 1 取 True（S_0 = S_T）；问题 2–4 取 False（跨日衔接，另查日界）。
    soc_init : 该日 0:00 的 SOC（问题 2–4 为跨日递推值；缺省用 cfg.SOC_INIT）。
    """
    T, dt = cfg.T, cfg.DT
    E, c, d, s, S = (sol[k] for k in ("E", "c", "d", "s", "SOC"))

    # 0. 维度断言（缺了它，全链路错成 T-1 期也会"通过"）
    assert len(E) == len(c) == len(d) == len(s) == T, "决策向量长度必须 = T"
    assert len(S) == T + 1, "SOC 必须含 T+1 个状态（0:00 … 24:00）"
    if g is None:
        g = np.zeros(T)
    else:
        assert len(g) == T, "紧急购电向量长度必须 = T"
    S0 = cfg.SOC_INIT if soc_init is None else float(soc_init)

    # 1. 原始平衡残差（独立重算，不复用求解器内部量；含紧急购电项）
    resid = E + PV*dt + d + g - (L*dt + c + s)
    assert np.abs(resid).max() <= tol, f"[{name}] 平衡残差 {np.abs(resid).max():.2e}"

    # 2. SOC 边界与递推
    assert S.min() >= cfg.SOC_LO - tol and S.max() <= cfg.SOC_HI + tol, \
        f"[{name}] SOC 越界 [{S.min()}, {S.max()}]"
    S_rec = S0 + np.cumsum(cfg.ETA_C*c - d/cfg.ETA_D)   # 放电效率在分母
    assert np.abs(S_rec - S[1:]).max() <= 1e-6, f"[{name}] SOC 递推不一致"

    # 3. 互斥性（效率<1 时最优解不会同时充放；见 02 §5）
    assert (c*d <= tol).all(), f"[{name}] 出现同时充放电"

    # 4. 功率上限（注意：变量是电量，故上限 = P_MAX·Δt）
    assert c.max() <= cfg.P_MAX*cfg.DT + tol and d.max() <= cfg.P_MAX*cfg.DT + tol

    # 5. 费用独立复算（关键：不信任求解器目标值）。
    #    阈值按全库统一表：费用复算用**相对** 1e-4（不要用 1e-9 或"绝对 1e-6 元"）
    cost = float((P*E).sum())      # E 已是 kWh，故不乘 Δt
    if "cost" in sol:
        assert abs(cost - sol["cost"]) <= max(1e-6, 1e-4*abs(cost)), \
            f"[{name}] 费用复算不一致 {cost} vs {sol['cost']}"

    # 6. 终端条件：问题 1 要求日周期；问题 2–4 只查日界衔接（跨日另查）
    if periodic:
        assert abs(S[-1] - S0) <= 1e-6, f"[{name}] 日周期不满足 S_T={S[-1]} vs S_0={S0}"

    return dict(name=name, cost=cost, max_resid=float(np.abs(resid).max()),
                soc_range=(float(S.min()), float(S.max())),
                cut=float(s.sum()), emergency=float(g.sum()))
```

## §5 报童对照：理论临界分位数 vs 数值经验覆盖率

```python
# newsboy.py
import numpy as np

def alpha_star(k, p, c_o):
    """临界分位数 α* = c_u/(c_u+c_o)，c_u = (k-1)·p。
    c_o 有三种口径，必须显式选择并说明（见 03 §1.2）：
      O1 过剩全浪费      c_o = p
      O2 可回收比例 θ    c_o = (1-θ)·p
      O3 储能吸纳        c_o = p − η·p_future
    """
    c_u = (k - 1.0)*p
    return c_u/(c_u + c_o)

def compare_with_numeric(need_actual, plan, k=5.0, p=None):
    """need_actual/plan: 逐时段的实际所需电量 / 计划购电量（1-D 等长数组）。

    ⚠️ 两个口径必须分清：
      - **服务水平** α = P(需求 ≤ 计划) = mean(plan >= need)  ← 这才是临界分位数 α* 的对拍量
      - **填充率** = Σmin(plan,need)/Σneed                     ← 只能作为辅助指标
    旧版用 mean(min(plan,need)/max(need,1e-9)) 是"量的填充率"，与 α* 不同量纲；
    且当 need<0（光伏盈余时段）时会给出负值（实测 emp ≈ −4.3e10）。
    """
    need = np.asarray(need_actual, float)
    plan = np.asarray(plan, float)
    assert need.shape == plan.shape, "need 与 plan 必须等长"
    pos = need > 0                                   # 只在"确有需求"的时段判定
    emp = float(np.mean(plan[pos] >= need[pos])) if pos.any() else float("nan")
    fill = float(np.minimum(plan[pos], need[pos]).sum() / need[pos].sum()) if pos.any() else float("nan")
    if p is None: p = 0.7
    a_O1 = alpha_star(k, p, p)                       # 过剩全浪费
    theta = 0.5                                       # 假设的回收比例
    a_O2 = alpha_star(k, p, (1-theta)*p)             # 部分回收
    return dict(service_level=emp, fill_ratio=fill, n_pos=int(pos.sum()),
                alpha_O1=a_O1, alpha_O2=a_O2,
                within=bool(min(a_O1, a_O2) - 0.05 <= emp <= max(a_O1, a_O2) + 0.05))
```

> ⚠️ **论文里必须写明这三条边界**：① 报童为**近似刻画**，非精确解；
> ② 多时段耦合与储能缓冲会使实际覆盖率偏离单时段值；
> ③ 分位数估计本身有样本误差，需给置信区间。

## §6 信息价值（VoI）对照：四组 + 频次扫描

```python
# voi.py
import numpy as np

def voi_table(run_model, L, PV, P, fc, cfg=None):
    """四层信息结构的费用对照。run_model 是你的求解入口，签名：
         run_model(L, PV, P, mode=..., fc=...) -> float(总费用)
    mode: 'PI' 完美信息 | 'TS' 两阶段(仅0:00) | 'MPC' 多阶段滚动 | 'OL' 开环
    ⚠️ 全部使用同一条真实轨迹 (L, PV, P)，只改"决策时知道什么"。"""
    out = {}
    out["G1_完美信息_PI"]  = run_model(L, PV, P, mode="PI",  fc=None)
    out["G2_两阶段_TS"]    = run_model(L, PV, P, mode="TS",  fc=fc)
    out["G3_多阶段_MPC"]   = run_model(L, PV, P, mode="MPC", fc=fc)
    out["G4_开环_OL"]      = run_model(L, PV, P, mode="OL",  fc=fc)
    # 自检不等式链（见 04 §3.1）：C(PI) ≤ C(MPC) ≤ C(TS) ≤ C(OL)
    # ⚠️ 只作**警告**、不要 assert：当模型含"调整成本"时（问题 3 正是如此），
    #    预报加密并不单调带来收益（见 03 §4.3），链条可以合法地被违反。
    #    另：mode='PI'/'TS'/'MPC'/'OL' 四种模型必须先在本文件中实现，否则本函数不可运行。
    chain = [out["G1_完美信息_PI"], out["G3_多阶段_MPC"],
             out["G2_两阶段_TS"],   out["G4_开环_OL"]]
    violation = [i for i in range(3) if chain[i] > chain[i+1]*(1+1e-6)]
    if violation:
        print(f"⚠️ 信息结构不等式链在位置 {violation} 被违反: {chain}"
              f"（若含调整成本，属预期现象，须在论文中解释）")
    out["VoI_滚动调整收益%"]   = 100*(out["G4_开环_OL"]-out["G3_多阶段_MPC"])/out["G4_开环_OL"]
    out["预测误差成本%"]       = 100*(out["G3_多阶段_MPC"]-out["G1_完美信息_PI"])/out["G1_完美信息_PI"]
    out["多引入3次预报收益%"]  = 100*(out["G2_两阶段_TS"]-out["G3_多阶段_MPC"])/out["G2_两阶段_TS"]
    return out

def voi_frequency_curve(run_model, L, PV, P, fc, ns=(1, 2, 4, 8, 24)):
    """预报频次 n 的边际收益曲线（直接回答题面"是否需要引入其他时刻的预报"）。

    ⚠️ **附件 3 只有 0:00/6:00/12:00/18:00 四个发布时刻**：n=1/2/4 可直接取子集，
       n=8/24 需 3:00/9:00/15:00/21:00 或每小时 —— 数据里**不存在**。
       因此 n>4 必须显式声明为**合成预报实验**并给出合成规则
       （如：在时刻 m 取最近的真实发布 m'≤m；或按提前期经验标准差对预报加噪降质），
       否则该组对照不可复现。
    做法：把可用预报时刻按 n 均匀取子集，重跑 G3。"""
    costs = {}
    for n in ns:
        hours = np.linspace(0, 24, n, endpoint=False).astype(int)
        costs[n] = run_model(L, PV, P, mode="MPC", fc=fc, issue_hours=tuple(hours))
    base = costs[1]
    marg = {n: (costs[ns[i-1]] - costs[n]) for i, n in enumerate(ns) if i > 0}
    return dict(costs=costs, marginal=marg, base=base)
```

```python
# ablation.py
import numpy as np, json

def run_ablation(run_variant, scen_path="intermediate/scenarios.npz"):
    """run_variant(name, scenarios) -> dict(费用, 紧急, 弃光, 违约, 时间)
    scenarios 必须先固定并存盘，所有变体读同一份 —— 这是公平性铁律。"""
    omega = np.load(scen_path)                      # 同一场景集，全部变体共用
    layers = [
        ("B0 无储能(闭式基线)",     "no_battery"),
        ("B1 储能+贪心策略",        "greedy"),
        ("B2 单日独立LP(清零边界)",  "daily_lp"),
        ("B3 +跨日SOC递推",         "cross_day"),
        ("B4 +弃光变量",            "curtail"),
        ("B5 +终端水价值",          "terminal_value"),
        ("B6 +不确定性(两阶段)",     "two_stage"),
        ("B7 +多阶段滚动",          "mpc"),
        ("M  完整模型(+波动电价)",   "full"),
    ]
    rows = []
    for label, key in layers:
        r = run_variant(key, omega)
        rows.append(dict(layer=label, **r))
    base = rows[0]["费用"]
    for i, r in enumerate(rows):
        r["相对B0%"] = 100*(r["费用"] - base)/base
        r["边际贡献%"] = 0.0 if i == 0 else 100*(rows[i-1]["费用"] - r["费用"])/base
    # 诚实检查：若某层边际贡献 > -0.5%（几乎无改善），必须在论文中说明其价值在别处
    for r in rows[1:]:
        if r["边际贡献%"] > -0.5:
            print(f"⚠️ {r['layer']} 边际贡献仅 {r['边际贡献%']:.2f}% —— "
                  f"请在论文中说明其价值在可行性/可解释性而非费用")
    return rows
```

## §8 全局敏感性（Sobol 指数）

```python
# sensitivity.py
import numpy as np
from scipy.stats import qmc

def sobol_indices(run_model, param_spec, L, PV, P, n_base=1024, seed=0):
    """param_spec: {name: (lo, hi)}；run_model(params)->float。
    使用 Saltelli 采样估计一阶指数 S_i（足够论文用；如需总效应 S_Ti 请用 SALib）。"""
    names = list(param_spec)
    d = len(names); rng = np.random.default_rng(seed)
    sampler = qmc.Sobol(d=2*d, scramble=True, seed=seed)
    base = sampler.random_base2(int(np.log2(n_base)))       # (n, 2d)
    A, B = base[:, :d], base[:, d:]
    lo = np.array([param_spec[n][0] for n in names])
    hi = np.array([param_spec[n][1] for n in names])
    scale = lambda U: lo + U*(hi - lo)
    fA = np.array([run_model(dict(zip(names, scale(u)))) for u in A])
    fB = np.array([run_model(dict(zip(names, scale(u)))) for u in B])
    varY = np.var(np.r_[fA, fB], ddof=1)
    S = {}
    for i, n in enumerate(names):
        AB = A.copy(); AB[:, i] = B[:, i]
        fAB = np.array([run_model(dict(zip(names, scale(u)))) for u in AB])
        S[n] = float((np.mean(fB*fAB) - np.mean(fA)*np.mean(fB)) / varY)
    return dict(S1=S, total_var=float(varY))
```

> **"结论在哪些参数范围内保持不变"**。
> **若所有参数的 $S_i$ 都很小，那也是一个好结论：说明模型稳健。**

```python
# audit_paper.py
import json
def audit(paper_numbers: dict, recomputed: dict, rel_tol=1e-4):
    """paper_numbers: 你从论文里手工抄下来的关键数字
       recomputed:    代码重算结果（results_summary.json）
       不一致就抛异常 —— 这是抓"改完代码忘改论文"的唯一可靠办法。"""
    bad = []
    for k, v in paper_numbers.items():
        if k not in recomputed:
            bad.append((k, v, "代码中无此指标")); continue
        r = recomputed[k]
        if abs(r - v) > max(1e-6, rel_tol*abs(v)):
            bad.append((k, v, r))
    if bad:
        raise AssertionError("论文与代码不一致：\n" + "\n".join(map(str, bad)))
    print(f"✅ {len(paper_numbers)} 个关键数字全部与代码一致")

if __name__ == "__main__":
    audit(json.load(open("paper_numbers.json")),
```
