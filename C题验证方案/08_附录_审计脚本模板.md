## 3. 问题 1 基准 LP（含/不含弃光）

```python
from scipy.optimize import linprog

def solve_q1(price, load, pv, with_slack=True, S0=S0, eta=ETA):
    """变量: E(n), Pc(n), Pd(n), S(n+1) [, s(n)]"""
    n = len(price); ns = n if with_slack else 0
    Nv = 4 * n + 1 + ns
    c = np.zeros(Nv); c[:n] = price                     # E 是 kWh，目标不用再乘 dt
    Aeq = np.zeros((2 * n + 1, Nv)); beq = np.zeros(2 * n + 1)

    for t in range(n):
        # 平衡: E + PV*dt + Pd*dt - Pc*dt (- s*dt) = L*dt
        Aeq[t, t] = 1.0
        Aeq[t, 2 * n + t] = DT
        Aeq[t, n + t] = -DT
        if with_slack:
            Aeq[t, 4 * n + 1 + t] = -DT
        beq[t] = (load[t] - pv[t]) * DT

        # SOC: S_{t+1} - S_t - eta*Pc*dt + Pd*dt/eta = 0
        r = n + t
        Aeq[r, 3 * n + t + 1] = 1.0
        Aeq[r, 3 * n + t] = -1.0
        Aeq[r, n + t] = -eta * DT
        Aeq[r, 2 * n + t] = DT / eta

    # 周期: S_0 - S_n = 0
    Aeq[2 * n, 3 * n] = 1.0
    Aeq[2 * n, 4 * n] = -1.0

    bounds = ([(0, None)] * n + [(0, PMAX)] * n + [(0, PMAX)] * n
              + [(SMIN, SMAX)] * (n + 1) + [(0, None)] * ns)
    bounds[3 * n] = (S0, S0)

    res = linprog(c, A_eq=Aeq, b_eq=beq, bounds=bounds, method="highs")
    assert res.success, res.message
    return res

# 典型日
```

## 3.5 结果文件：位置式读回与 4 h 块级审计

> 上面 §3 的 LP 解是**审计脚本自己**解出来的，用 `audit()` 审它等于**自证**（约束代数与常数共享）。
> 真正独立的一步是**读回 `result*.xlsx` 再复算**，但必须接受一条现实约束：
> **交付物里不存在 10 min 粒度的 P_c/P_d**（只有 6 个 4 h 聚合充放值 + 0:00/24:00 两个储电量），
> 因此 **10 min 级验算只能在模型自己的解上做，不得声称已核验交付物**；对交付物另设 4 h 块级审计。

### 版式与位置式读法（禁止按列名比对）

| 交付物 | 版式 | 末标签写法 |
|---|---|---|
| `result1`·计划购电量 | 145×2 竖表，**无**全天行 | `0:00+1-0:10+1` |
| `result2/3/4-*`·计划/调整购电量 | 335×147，**334 个日期行已预填**，末两列为"全天购电量/全天购电费" | `0:00-0:10+1` |
| `result2/4-2`·充放电量 | 20 行 = 2–3 个示例日 + 1 行字面量 `⁝` | — |
| `result3/4-3`·充放电量 | 26 行 = 同上 | — |
| 紧急购电量 | 11 行，含 `⁝` 行 | — |

- **对应关系写死（位置式）**：附件 5 第 j 期（j=0..143）↔ 附件 1 第 j 行 ↔ 附件 2/4 第 j+1 列；
- `0:00` 储电量落在 `0:00-4:00` 行、`24:00` 落在 `4:00-8:00` 行；
- 充放电量必须**删掉 `⁝` 行并重建 334×6 = 2 004 行**；紧急购电量按事件重建；
- 断言只用"工作表名 + 标签**计数**（147/145/144）+ `result2/3/4-*` 日期范围 2025-02-01…12-31（334 行）"，
  **不要**按列名逐一比对：附件 1 时间列含 60 个时间值 + 84 个字符串、附件 2 表头含 143 个时间值 + 2 个字符串，
  且模板末标签在 `result1` 与 `result2/3/4-*` 上写法不同、result2/3/4-* 第 43 个表头单元格还有笔误 `7:0-7:10`。

```python
def load_result(path, sheet, has_date=True):
    """
    位置式读回结果文件（不做任何列名匹配）。
      result1        : 145x2（144 期 + 表头）；末行标签 `0:00+1-0:10+1`；**无全天行**
      result2/3/4-*  : 335x147（334 个日期行 + 表头），第 0 列 = 日期，
                       第 1..144 列 = 各 10 min 时段，第 145/146 列 = 全天购电量 / 全天购电费
    返回 (labels, values, daily_totals)；标签/列数按位置断言，不做列名比对。
    """
    raw = pd.read_excel(path, sheet_name=sheet, header=None).to_numpy()
    n_rows, n_cols = raw.shape
    labels = raw[1:, 0].astype(str)              # result1 为时段标签；result2/3/4-* 为日期
    if has_date:
        assert (n_rows, n_cols) == (335, 147), (n_rows, n_cols)
        values = raw[1:, 1:145].astype(float)    # 334 x 144
        daily_totals = raw[1:, 145:147].astype(float)   # 334 x 2
        assert values.shape == (334, 144), values.shape
        assert str(labels[0])[:10] == "2025-02-01" and str(labels[-1])[:10] == "2025-12-31"
    else:
        assert (n_rows, n_cols) == (145, 2), (n_rows, n_cols)
        values = raw[1:, 1:2].astype(float)      # 144 x 1
        daily_totals = None
        assert str(labels[-1]) == "0:00+1-0:10+1", labels[-1]   # 末标签写法与 result2 不同
    return labels, values, daily_totals

def audit_4h_blocks(S_edges, E_b, Ec_b, Ed_b, g_b, PV_b, L_b, s_b,
                    eta=ETA, Smin=SMIN, Smax=SMAX, Pmax=PMAX, tol=1e-6):
    """
    4 h 块级审计（b = 0..5）——交付物上唯一可复核的粒度。
    块内所有量均为 4 h 电量 [kWh]：PV_b = Σ_{t∈b} PV_t·Δt，L_b、s_b 同理；
    S_edges = 7 个时刻（0:00, 4:00, ..., 24:00）的储电量（从"充放电量"表按 4 h 行读取，
    其中 0:00 在 `0:00-4:00` 行、24:00 在 `4:00-8:00` 行）。
      1) SOC 块递推 : S_{b+1} - S_b - eta*Ec_b + Ed_b/eta = 0
      2) 块内平衡   : E_b + g_b + PV_b + Ed_b - L_b - Ec_b - s_b = 0
      3) 边界       : S_b ∈ [Smin, Smax]、Ec_b/Ed_b ∈ [0, 4*Pmax]
    """
    S_edges = np.asarray(S_edges, float)
    soc_res = S_edges[1:] - S_edges[:-1] - eta * Ec_b + Ed_b / eta
    bal_res = E_b + g_b + PV_b + Ed_b - L_b - Ec_b - s_b
    out = {
        "block_soc_res": float(np.abs(soc_res).max()),
        "block_balance_res": float(np.abs(bal_res).max()),
        "block_soc_violations": int(((S_edges < Smin - 1e-6) | (S_edges > Smax + 1e-6)).sum()),
        "block_power_violations": int(((Ec_b < -1e-6) | (Ec_b > 4 * Pmax + 1e-6)
                                       | (Ed_b < -1e-6) | (Ed_b > 4 * Pmax + 1e-6)).sum()),
    }
    assert out["block_soc_res"] <= tol and out["block_balance_res"] <= tol, out
    assert out["block_soc_violations"] == 0 and out["block_power_violations"] == 0, out
    return out

# 注：10 min 级 SOC/平衡递推（§4 的 audit）只能在**模型自己的解**上做；
#     对交付物只能跑上面的块级审计，**不得声称已核验交付物**。
```

## 4. 通用审计函数（解向量 → 物理量复算）

```python
def audit(price, load, pv, E, Pc, Pd, s, S, g=None, dt=DT, eta=ETA,
          Smin=SMIN, Smax=SMAX, Pmax=PMAX, cyclic=True, require_g=False):
    """
    E, Pc, Pd, s: shape (n,)   E[kWh], Pc/Pd/s[kW]
    S: shape (n+1,) [kWh]
    g: shape (n,) 紧急购电量 [kWh]。**问题 2/3/4-3 必填**（require_g=True）：
       附件 5 把"计划购电量"与"紧急购电量"分成两张表，计划量 ≠ 实际供给量；
       漏掉 g 时平衡残差恰等于 g（实测合法结算 g[42]=600 kWh ⇒ 残差 600.000 kWh，
       超 1e-6 容差 6.0e8 倍），补 +g 后为 1.49e-13。问题 1 时 g≡0。
    返回残差字典；任一超阈值即视为失败。
    """
    n = len(price)
    assert n == 144, f"期望 144 期（10 min），实得 {n} 期——错成 143 期会静默通过"
    assert len(S) == n + 1, f"S 应为 n+1 = {n+1} 个时刻，实得 {len(S)}"
    if require_g:
        assert g is not None, "问题 2/3/4-3 的紧急购电量 g 是必填参数（缺了它平衡式凭空少一项）"
    gv = np.zeros(n) if g is None else np.asarray(g, float)
    out = {}
    # 结算平衡必须含紧急购电：E + PV·Δt + Pd·Δt + g = L·Δt + Pc·Δt + s·Δt
    balance = E + pv * dt + Pd * dt + gv - load * dt - Pc * dt - s * dt
    soc = S[1:] - S[:-1] - eta * Pc * dt + Pd * dt / eta
    out["max_balance_res"] = float(np.abs(balance).max())
    out["max_soc_res"] = float(np.abs(soc).max())
    out["soc_violations"] = int(((S < Smin - 1e-6) | (S > Smax + 1e-6)).sum())
    out["power_violations"] = int(((Pc < -1e-6) | (Pc > Pmax + 1e-6)
                                   | (Pd < -1e-6) | (Pd > Pmax + 1e-6)).sum())
    out["max_simultaneous"] = float((Pc * Pd).max())
    out["nonneg"] = float(min(E.min(), Pc.min(), Pd.min(), s.min(), S.min()))
    if cyclic:
        out["cyclic_gap"] = float(abs(S[0] - S[-1]))
    out["emergency_negative"] = float(gv.min())          # g≡0（问题 1）时为 0
    out["emergency_nonneg"] = bool((gv >= -1e-6).all())  # 以前从不检查 g，必须补
    out["cost_plan"] = float((price * E).sum())          # 与 result*.xlsx"全天购电费"列对拍
    return out

def assert_pass(res, tol_balance=1e-6, tol_soc=1e-6, tol_mutex=1e-6):
    checks = {
        "balance": res["max_balance_res"] <= tol_balance,
        "soc": res["max_soc_res"] <= tol_soc,
        "soc_bounds": res["soc_violations"] == 0,
        "power_bounds": res["power_violations"] == 0,
        "mutex": res["max_simultaneous"] <= tol_mutex,
        "nonneg": res["nonneg"] >= -1e-6,
        "emergency_nonneg": res["emergency_nonneg"],
        "cyclic": res.get("cyclic_gap", 0.0) <= 1e-6,
    }
    print(checks)
    assert all(checks.values()), checks
    return checks
```

```python
def recompute_cost_q2(price, E_plan, g):
    """
    问题 2 口径（`03` 的 Q2-V3 要求此复算，原模板完全缺失）：
      总费用 = Σ p_t·E_plan_t + Σ 5·p_t·g_t
    口径声明：附件 5 把"计划购电量"与"紧急购电量"分成两张表，**计划量 ≠ 实际供给量**；
    计划费按 E_plan 全额计，供给缺口由 g 以 5 倍价补足（g 由附件 2 实际值结算得到）。
    """
    E_plan, g = map(np.asarray, (E_plan, g))
    return float((price * E_plan).sum() + (5.0 * price * g).sum())

# 与结果文件对拍：result2/3/4-* 的"全天购电费"列应等于 Σ p_t·E_plan_t（相对差 ≤1e-4）；
# 若该列已把紧急购电计入，则改用 Σp·E_plan + Σ5p·g，并在假设节声明该列口径。

def recompute_cost_q3(price, E_plan, E_adj, g):
    """
    统一口径：price[元/kWh]，E_plan/E_adj/g[时段 kWh]。
    计划费全额 + 调减违约金 0.5p·max(plan-adj,0) + 调增 1.5p·max(adj-plan,0) + 紧急 5p·g
    """
    E_plan, E_adj, g = map(np.asarray, (E_plan, E_adj, g))
    diff = E_adj - E_plan
    cost = (price * E_plan).sum()
    cost += (1.5 * price * np.maximum(diff, 0)).sum()
    cost += (0.5 * price * np.maximum(-diff, 0)).sum()
    cost += (5.0 * price * g).sum()
    return cost

# 若另一种读法为"按实际调整量付费 + 差额惩罚"，则改为：
#   cost = (price*E_adj).sum() + 0.5p·max(plan-adj,0) + 1.5p·max(adj-plan,0) + 5p·g
# 两种口径必须在假设节声明，并与主模型一致。
# 注：本题主口径下 E_adj < E_plan 的违约分支在最优解中永不活跃（成本对 E_adj 的偏导 = -0.5p < 0），
#     故"计划费全额 + 违约金"与"按实际调整量付费"两种读法在最优解上同值；该分支只作对照口径（`04` §6）。
```

## 6. 非预期性（信息泄漏）测试模板

```python
# 只对**浮点数组**做扰动：日期(datetime64)/索引(int)/字符串列一律原样返回，
# 否则 `arr[t:] = arr[t:] * 2 + 1` 会在 datetime64 上抛 UFuncTypeError。
NON_PERTURB_KEYS = {"dates", "issue_time", "day_index", "target_hours", "labels"}

def perturb(obj, t, keys=None):
    """
    递归扰动 t 之后的数据。
      - keys 给定时只扰动白名单内的键（如 {"load", "pv", "price", "forecast"}）；
      - 嵌套 dict（**预报表，正是 Q3 的泄漏通道**）会被递归深入处理，不再被静默跳过。
    """
    if isinstance(obj, dict):
        return {k: (perturb(v, t, keys)
                    if (k not in NON_PERTURB_KEYS and (keys is None or k in keys)) else v)
                for k, v in obj.items()}      # 保留全部键，只扰动允许的键
    arr = np.asarray(obj)
    if arr.ndim == 0 or not np.issubdtype(arr.dtype, np.floating):
        return obj                       # datetime64 / int 索引 / 字符串：跳过扰动
    out = arr.copy()
    out[t:] = out[t:] * 2.0 + 1.0        # 任意扰动
    return out

def leakage_test(policy_fn, data, t, keys=None):
    """
    policy_fn(data) -> decisions, 其中 decisions[t] 是 t 时刻决策。
    要求：扰动 t 之后的未来数据后，**t 及之前的决策逐位不变**。
    """
    x1 = policy_fn(data)
    x2 = policy_fn(perturb(data, t, keys))
    diff = np.abs(np.asarray(x1)[:t + 1] - np.asarray(x2)[:t + 1]).max()
    print(f"leakage max diff before t={t}: {diff:.2e}")
    assert diff <= 1e-6, "检测到信息泄漏：过去决策受未来数据影响"
    return diff

# ---- 阳性 / 阴性对照：没有阳性对照，常量策略也会 PASS，结论没有检验力 ----
def cheater_policy(data):
    """阳性对照：故意使用未来信息（决策直接依赖被扰动的未来价格/预报）。"""
    return np.asarray(data["price"], float).copy()

def constant_policy(data):
    """阴性对照：常量策略，任何扰动都不会改变它（应当 PASS）。"""
    return np.zeros_like(np.asarray(data["price"], float))

# 运行顺序（缺一不可）：
#   assert leakage_test(cheater_policy, data, t) > 1e-6   # 阳性对照**必须被检出**
#   leakage_test(policy_fn, data, t)                      # 待测策略必须 PASS
```

**必测时刻**：Q2 的 0:00；Q3 的 6:00/12:00/18:00；Q4 的对应决策时刻。

**测试方向（按 `04` §3.3–3.4、裁定基准 D7）**：

- 附件 3 的 18:00 发布的是**未来 24 小时**预报，**跨日部分的使用是合法的**，不是泄漏；
- 扰动对象只能选：① **更晚发布**的预报（测 t=6:00 时扰动 12:00/18:00 的预报）、② **未来实际值**（附件 2 的次日/当日更晚真值）；
- **必须保留"发布时刻 = t 的那条预报"整条不被扰动**——否则等于把策略的合法输入也改了，必然误报；
- Q2 的"0:00 计划"是**全天 144 维向量**（题面即"根据附件 1 的电价和附件 2 的数据"制定），
  扰动 0:00 之后的 L/PV 真值**必然**改变同日计划：按 R1 只做**跨日**测试（d 之前各日计划逐位不变），
  按 R3 只做**日内**测试（只扰动更晚发布的预报与未来实际值）。

## 8. 储能纯套利：循环次数受控的收益

**定义式（参数名/注释/实现必须一致）**：

> **等效循环次数 = 总放电量 / (η_c × 可用容量)**，可用容量 = S_max − S_min = 9 600 kWh。
> 因此"单循环" = 总放电量 **8 640 kWh** = η_c×9 600，"双循环" = **17 280 kWh**。

- `arbitrage_eq_total_discharge` 里的 `discharge_cap` 是**等式**约束（强制恰好放这么多电）：
  `None` → 无限制 11 769.31 元（充 26 500 / 放 21 465 kWh = **2.236** 次）；
  `8640.0` → "单循环" **7 256.20** 元；`17280.0` → "双循环" **11 321.04** 元；
- 若只要"总放电量**不超过**某值"的**上限**语义，用下面的 `arbitrage_ub_total_discharge`（`A_ub` 版本）。

```python
def arbitrage_eq_total_discharge(price, eta=0.9, S0=6000.0, Smin=1200.0, Smax=10800.0,
                                 Pmax=5000.0, dt=1/6, discharge_cap=None):
    """
    平均日电价下、周期 SOC、无 PV/负载的纯套利最大收益（日收益，元）。

    discharge_cap 是**总放电量**的**等式**约束：Σ_t Pd_t·dt = discharge_cap。
      等效循环次数 = 总放电量 / (eta * (Smax - Smin))：
        None    -> 无限制，放 21465 kWh ⇒ 2.236 次，收益 11769.31 元
        8640.0  -> 单循环（= eta * 9600），收益 7256.20 元
        17280.0 -> 双循环，收益 11321.04 元
    """
    n = len(price); Nv = 3 * n + 1
    c = np.zeros(Nv); c[:n] = price * dt; c[n:2*n] = -price * dt
    Aeq = np.zeros((n + 2, Nv)); beq = np.zeros(n + 2)
    for t in range(n):
        # S_{t+1} - S_t - eta*Pc*dt + Pd*dt/eta = 0
        Aeq[t, 2*n + t + 1] = 1.0; Aeq[t, 2*n + t] = -1.0
        Aeq[t, t] = -eta * dt; Aeq[t, n + t] = dt / eta
    # 周期
    Aeq[n, 2*n] = 1.0; Aeq[n, 3*n] = -1.0
    # 总放电量**恰好等于** discharge_cap（等式语义，见 docstring）
    if discharge_cap is not None:
        Aeq[n + 1, n:2*n] = dt
        beq[n + 1] = discharge_cap
    bounds = ([(0, Pmax)] * n + [(0, Pmax)] * n + [(Smin, Smax)] * (n + 1))
    bounds[2*n] = (S0, S0)
    res = linprog(c, A_eq=Aeq, b_eq=beq, bounds=bounds, method="highs")
    assert res.success, (f"infeasible: discharge_cap={discharge_cap} 超过可达最大值"
                         f"（受 {Pmax} kW 功率界与周期 SOC 约束限制）")
    return -res.fun

def arbitrage_ub_total_discharge(price, eta=0.9, S0=6000.0, Smin=1200.0, Smax=10800.0,
                                 Pmax=5000.0, dt=1/6, discharge_cap=8640.0):
    """**上限**语义（A_ub）：Σ_t Pd_t·dt ≤ discharge_cap，即"至多 k 次循环"的收益上界。"""
    n = len(price); Nv = 3 * n + 1
    c = np.zeros(Nv); c[:n] = price * dt; c[n:2*n] = -price * dt
    Aeq = np.zeros((n + 1, Nv)); beq = np.zeros(n + 1)
    for t in range(n):
        Aeq[t, 2*n + t + 1] = 1.0; Aeq[t, 2*n + t] = -1.0
        Aeq[t, t] = -eta * dt; Aeq[t, n + t] = dt / eta
    Aeq[n, 2*n] = 1.0; Aeq[n, 3*n] = -1.0
    Aub = np.zeros((1, Nv)); Aub[0, n:2*n] = dt
    bounds = ([(0, Pmax)] * n + [(0, Pmax)] * n + [(Smin, Smax)] * (n + 1))
    bounds[2*n] = (S0, S0)
    res = linprog(c, A_eq=Aeq, b_eq=beq, A_ub=Aub, b_ub=[discharge_cap],
                  bounds=bounds, method="highs")
    assert res.success, f"infeasible: discharge_cap={discharge_cap} 参数有误"
    return -res.fun

# 参考（实测）: 单循环 7256.20、双循环 11321.04、无限制 11769.31 元/日
#（平均日电价口径；"单/双循环"= 总放电量固定 8640 / 17280 kWh 的等式约束）
```

## 9. 预报误差：配对比较（禁止跨样本比均值）

```python
# day_index: 日索引映射（原模板未定义，必须先建）
day_index = {str(d.date()): i for i, d in enumerate(pd.date_range("2025-01-01", "2025-12-31"))}

def forecast_error_pairs(F, mt, fdates, PV, day_index, m_issue, target_hours):
    """
    返回给定发布时刻 m_issue、给定目标整点集合 target_hours 的 (error, actual) 数组。

    对齐（**绝对时间轴**，允许跨日）：发布日 d0 的第 k 条预报
        目标绝对整点 g = d0*24 + m_issue + k − 1   （k = 1..24）
    即"预报 k 小时 = [h0+k−1, h0+k)"。18:00 发布的 k=7..24 落在**次日** 0:00–17:00，
    必须按 g 映射到次日；若强行取同日 PVh[d, h]，会**静默丢弃 18:00 的次日时域**
    ——这正是样本量崩到 121 的原因。
    越界配对（g 超出 365×24）丢弃并**报告丢弃数**：实测 36/35 040 = 0.103%。
    """
    PVh = np.array([PV[:, 6*h:6*h+6].mean(1) for h in range(24)]).T   # 365x24 逐小时均值
    fdates = pd.to_datetime(pd.Series(fdates).reset_index(drop=True))  # 保证可按位置取
    # 括号写清：`mt == X if cond else mt == Y` 是优先级陷阱（原先写法靠猜）
    rows = np.where((mt == f"{m_issue}:00") if m_issue > 0 else (mt == "0:00"))[0]
    H = PVh.shape[0] * 24                                            # 总目标整点数 = 365*24
    errs, acts, dropped, tried = [], [], 0, 0
    for r in rows:
        d0 = day_index[str(fdates[r].date())]
        for k in range(1, 25):
            tried += 1
            g = d0 * 24 + m_issue + k - 1                            # 绝对目标整点（可跨日）
            if g >= H:
                dropped += 1
                continue
            if (g % 24) in target_hours:
                errs.append(F[r, k - 1] - PVh[g // 24, g % 24])
                acts.append(PVh[g // 24, g % 24])
    print(f"dropped {dropped}/{tried} = {dropped / max(tried, 1):.3%}")
    return np.array(errs), np.array(acts)

# 配对示例: 0:00 发布 vs 12:00 发布，共同目标整点 12–18 时（≈lead 16 h vs 4 h）
# 期望: MAE 697.1 vs 540.7（同一批目标整点, 比值 1.29）
# 注意: 1.29× **不是** 6h→24h 的替代值——受控配对的 6h→24h 为 2.01×（全样本）/ 2.03×（白天）。
```
